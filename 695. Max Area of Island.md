### Step1

前回の問題とほぼ同じ。前回は島の数を求めたが、今回は島の陸(1)が最大のものを出力する。
入力のサイズがかなり小さいので、実行時間は問題なさそうなので、パフォーマンスよりも可読性に気をつけたい。
前回のコメントから、このグラフの探索ではDFSの方がBFSより省メモリではないことを知ったので、BFSでも書いてみる。
再帰かスタックどちらを使うかだが、スタックの方がカウントしていく処理が直感的になりそうなので、スタックを選択した。


```cpp
class Solution {
    const static int kWater = 0;
    const static int kLand = 1;
    const static int kVisited = -1;

    int count_area(int s_row, int s_col, std::vector<std::vector<int>>& grid) {
        const int num_rows = grid.size();
        const int num_cols = grid[0].size();

        int area = 0;
        std::stack<std::pair<int, int>> frontiers;
        frontiers.emplace(s_row, s_col);
        while (!frontiers.empty()) {
            auto [row, col] = frontiers.top();
            frontiers.pop();
            if (!(0 <= row && row < num_rows)) {
                continue;
            }
            if (!(0 <= col && col < num_cols)) {
                continue;
            }
            if (grid[row][col] == kWater || grid[row][col] == kVisited) {
                continue;
            }
            grid[row][col] = kVisited;
            ++area;
            frontiers.emplace(row - 1, col);
            frontiers.emplace(row, col - 1);
            frontiers.emplace(row, col + 1);
            frontiers.emplace(row + 1, col);
        }
        return area;
    }

public:
    // WARNING: grid will be modified by changing all 1's to -1's
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        const int num_rows = grid.size();
        const int num_cols = grid[0].size();
        int max_area = 0;
        for (int row = 0; row < num_rows; ++row) {
            for (int col = 0; col < num_cols; ++col) {
                if (grid[row][col] == kLand) {
                    int area = count_area(row, col, grid);
                    max_area = std::max(max_area, area);
                }
            }
        }
        return max_area;
    }
};
```

class Solution {
    static const int kLand = 1;
    static const int kWater = 0;
    static const int kVisited = -1;

    int count_area(int row, int col, vector<vector<int>>& grid) {
        const int num_rows = grid.size();
        const int num_cols = grid[0].size();

        if (!(0 <= row && row < num_rows)) {
            return 0;
        }
        if (!(0 <= col && col < num_cols)) {
            return 0;
        }
        if (grid[row][col] != kLand) {
            return 0;
        }
        int area = 1;
        grid[row][col] = kVisited;
        area += count_area(row - 1, col, grid);
        area += count_area(row, col - 1, grid);
        area += count_area(row, col + 1, grid);
        area += count_area(row + 1, col, grid);
        return area;
    }

public:
    int maxAreaOfIsland(vector<vector<int>>& grid) {
        const int num_rows = grid.size();
        const int num_cols = grid[0].size();

        int max_area = 0;
        for (int row = 0; row < num_rows; ++row) {
            for (int col = 0; col < num_cols; ++col) {
                if (grid[row][col] == kLand) {
                    int area = count_area(row, col, grid);
                    max_area = std::max(max_area, area);
                }
            }
        }
        return max_area;
    }
};

### Union Find の解法

参考：https://ja.wikipedia.org/wiki/%E7%B4%A0%E9%9B%86%E5%90%88%E3%83%87%E3%83%BC%E3%82%BF%E6%A7%8B%E9%80%A0

```cpp
#include <algorithm>
#include <vector>

class UnionFind {
  std::vector<int> parent_;
  std::vector<int> size_;

public:
  UnionFind(std::size_t max_size);
  int find_set(int x);
  int union_set(int x, int y);
  int get_size(int x);
};

UnionFind::UnionFind(std::size_t max_size) {
  parent_.resize(max_size);
  for (int i = 0; i < max_size; ++i) {
    parent_[i] = i;
  }
  size_.resize(max_size, 1);
}

int UnionFind::find_set(int x) {
  if (parent_[x] == x) {
    return x;
  }
  parent_[x] = find_set(parent_[x]);
  return parent_[x];
}

int UnionFind::union_set(int x, int y) {
  int x_root = find_set(x);
  int y_root = find_set(y);
  if (x_root == y_root) {
    return x_root;
  }
  int smaller = x_root;
  int bigger = y_root;
  if (size_[x_root] > size_[y_root]) {
    smaller = y_root;
    bigger = x_root;
  }
  parent_[smaller] = bigger;
  size_[bigger] += size_[smaller];
  return bigger;
}

int UnionFind::get_size(int x) { return size_[find_set(x)]; }

// ---------------------------------------------------------
// ここからが Max Area of Island
// ---------------------------------------------------------

class Solution {
public:
  int maxAreaOfIsland(std::vector<std::vector<int>> &grid) {
    const int num_rows = grid.size();
    const int num_cols = grid[0].size();

    UnionFind union_find(num_rows * num_cols);
    int max_area = 0;
    for (int row = 0; row < num_rows; ++row) {
      for (int col = 0; col < num_cols; ++col) {
        if (grid[row][col] == 0) {
          continue;
        }
        max_area = std::max(max_area, 1);
        int id = row * num_cols + col;
        if (col + 1 < num_cols && grid[row][col + 1] == 1) {
          int root = union_find.union_set(id, row * num_cols + col + 1);
          max_area = std::max(max_area, union_find.get_size(root));
        }
        if (row + 1 < num_rows && grid[row + 1][col] == 1) {
          int root = union_find.union_set(id, (row + 1) * num_cols + col);
          max_area = std::max(max_area, union_find.get_size(root));
        }
      }
    }
    return max_area;
  }
};
```
