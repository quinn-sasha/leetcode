### Step1

おそらく解いたことがあったので、解法自体は思いつくのに時間はかからなかった。

思考過程：

2次元配列の要素を一つづつ見ていって、未到達の陸なら、そこから島全体を探索して既に訪れた場所を記録していく。
まずは探索方法を比較した。DFS、 BFSどちらでも動くが、今回の問題では全ての島を訪れることだけが目的であり、BFSを使うと必要以上にメモリを多く消費することになってしまう。だからDFSを選択した。
再帰で書くかスタックで書くか悩んだが、入力のサイズを考えるとそこまでメモリの消費は多くなさそうなので書きやすそうな再帰を選択した。
問題は訪れたことをどのように記録するかだが、最初は std::set に行と列のペアを保存した。しかし、この方法だとsetを参照、追加するさいに O(log MN) (M: 行数、N: 列数）のコストがかかってしまうので、最終的な時間計算量は O(MN log MN) になってしまう。
‘1’ を ‘0’ に書き換えていく方法であれば追加のデータ構造なしで O(1) でアクセスできるが、入力を書き換えてしまって良いのかどうかはわからない。
よって、追加でデータ構造は必要だが、O(1)でアクセスできる、入力と同じサイズのboolの2次元配列を選択した。これで時間計算量は O(MN) になった。

```cpp
class Solution {
    void traverse(int s_row, int s_col,
                  int num_rows, int num_cols,
                  std::vector<std::vector<bool>>& visited,
                  std::vector<std::vector<char>>& grid) {
        std::vector<std::pair<int, int>> adjacents = {
            {s_row - 1, s_col}, {s_row, s_col - 1}, 
            {s_row, s_col + 1}, {s_row + 1, s_col}
        };
        for (auto [row, col] : adjacents) {
            if (row < 0 || col < 0) {
                continue;
            }
            if (row >= num_rows || col >= num_cols) {
                continue;
            }
            if (grid[row][col] == '0') {
                continue;
            }
            if (visited[row][col]) {
                continue;
            }
            visited[row][col] = true;
            traverse(row, col, num_rows, num_cols, visited, grid);
        }
    }

public:
    int numIslands(vector<vector<char>>& grid) {
        int num_rows = grid.size();
        int num_cols = grid[0].size();
        std::vector<std::vector<bool>> visited(
            num_rows, std::vector<bool>(num_cols, false)
        );
        int num_islands = 0;
        for (int row = 0; row < num_rows; row++) {
            for (int col = 0; col < num_cols; col++) {
                if (grid[row][col] == '1' && !visited[row][col]) {
                    visited[row][col] = true;
                    traverse(row, col, num_rows, num_cols, visited, grid);
                    num_islands++;
                }
            }
        }
        return num_islands;
    }
};
```
### Step2

他の人のコードを読み、自分のコードを振り返る：
再帰の書き方は、traverse()する前に訪れたことを記録するか、traverse()の中で記録するかでも書き方を変えられることに気づいた。
あと関数の引数が多すぎると思った。num_rows、num_cols を関数の引数で持ち運ぶよりも、girdから取得する方がいい。

毎回 vector を作るのは確かに重い処理だと思った。スタックメモリを使おう。
https://github.com/dxxsxsxkx/leetcode/pull/17#discussion_r2693621301

isOutOfBounds()で表から飛び出していないかチェックするのがいいと思った。
スタックでの解法は、以下のコードを参考にした。

```cpp
class Solution {
    static const char kWater = '0';
    static const char kLand = '1';
    static const char kVisited = '*';

    void mark_lands_visited(int s_row, int s_col, 
                            std::vector<std::vector<char>>& grid) {
        const int num_rows = grid.size();
        const int num_cols = grid[0].size();

        std::stack<std::pair<int, int>> frontiers;
        frontiers.emplace(s_row, s_col);
        while (!frontiers.empty()) {
            auto [row, col] = frontiers.top();
            frontiers.pop();
            if (!(0 <= row && row < num_rows && 0 <= col && col < num_cols)) {
                continue;
            }
            if (grid[row][col] == kWater || grid[row][col] == kVisited) {
                continue;
            }
            grid[row][col] = kVisited;
            frontiers.emplace(row - 1, col);
            frontiers.emplace(row, col - 1);
            frontiers.emplace(row, col + 1);
            frontiers.emplace(row + 1, col);
        }
    }

public:
    // WARNING: grid will be modified by changing all '1's to '*'s
    int numIslands(vector<vector<char>>& grid) {
        int num_islands = 0;
        for (int row = 0; row < grid.size(); ++row) {
            for (int col = 0; col < grid[0].size(); ++col) {
                if (grid[row][col] == kLand) {
                    mark_lands_visited(row, col, grid);
                    ++num_islands;
                }
            }
        }
        // You can restore original map by changing '1' to '*'
        return num_islands;
    }
};
```

### Step3

最終的には再帰でgridを破壊していく方法が書きやすいことがわかった。

```cpp
class Solution {
    const static char kWater = '0';
    const static char kLand = '1';
    const static char kVisited = '*';

    void traverse(int row, int col, std::vector<std::vector<char>>& grid) {
        const int num_rows = grid.size();
        const int num_cols = grid[0].size();

        if (!(0 <= row && row < num_rows)) {
            return;
        }
        if (!(0 <= col && col < num_cols)) {
            return;
        }
        if (grid[row][col] == kWater || grid[row][col] == kVisited) {
            return;
        }
        grid[row][col] = kVisited;
        traverse(row - 1, col, grid);
        traverse(row, col - 1, grid);
        traverse(row, col + 1, grid);
        traverse(row + 1, col, grid);
    }

public:
    // WARNING: grind will be modified by changing all '1's to '*'s
    int numIslands(vector<vector<char>>& grid) {
        const int num_rows = grid.size();
        const int num_cols = grid[0].size();
        
        int num_islands = 0;
        for (int row = 0; row < num_rows; ++row) {
            for (int col = 0; col < num_cols; ++col) {
                if (grid[row][col] == kLand) {
                    traverse(row, col, grid);
                    ++num_islands;
                }
            }
        }
        // You can restore original map by changing '*'s to '1's
        return num_islands;
    }
};
```
