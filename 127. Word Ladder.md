```python
class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        if endWord not in wordList:
            return 0

        pattern_to_words = defaultdict(list)
        wordList.append(beginWord)
        for word in wordList:
            for pos in range(len(word)):
                pattern = word[:pos] + "*" + word[pos + 1:]
                pattern_to_words[pattern].append(word)
        
        seen_words = set([beginWord])
        current_words = deque([beginWord])
        count = 1
        while current_words:
            for _ in range(len(current_words)):
                word = current_words.popleft()
                if word == endWord:
                    return count
                for pos in range(len(word)):
                    pattern = word[:pos] + "*" + word[pos + 1:]
                    for next_word in pattern_to_words[pattern]:
                        if next_word in seen_words:
                            continue
                        seen_words.add(next_word)
                        current_words.append(next_word)
            count += 1
        return 0
```

```python
class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        candinates = set(wordList)
        if not endWord in candinates:
            return 0
        
        visited = set([beginWord])
        def get_adjacent_word(word):
            def is_adjacent(word1, word2):
                distance = 0
                for i in range(len(word1)):
                    if word1[i] != word2[i]:
                        distance += 1
                    if distance > 1:
                        return False
                return distance == 1

            for candinate in candinates:
                if not is_adjacent(word, candinate):
                    continue
                if candinate in visited:
                    continue
                visited.add(candinate)
                yield candinate
    
        path_length = 1
        words = [beginWord]
        while words:
            next_words = []
            for word in words:
                if word == endWord:
                    return path_length
                for adjacent_word in get_adjacent_word(word):
                    next_words.append(adjacent_word)
            words = next_words
            path_length += 1
        return 0
```


- 上記の場合、隣接する単語を探すごとに、入力を走査している
- Adjacency list を作って改善できる

```python
class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        if not endWord in wordList:
            return 0

        def is_adjacent(word1, word2):
            distance = 0
            for i in range(len(word1)):
                if word1[i] != word2[i]:
                    distance += 1
                if distance > 1:
                    return False
            return distance == 1

        wordList.append(beginWord)  # To prevent KeyError (adjacents[word])
        adjacents = {word: set() for word in wordList}
        for i in range(len(wordList)):
            for j in range(i + 1, len(wordList)):
                if not is_adjacent(wordList[i], wordList[j]):
                    continue
                adjacents[wordList[i]].add(wordList[j])
                adjacents[wordList[j]].add(wordList[i])
        
        visited = set([beginWord])
        words_and_length = deque([(beginWord, 1)])
        while words_and_length:
            word, path_length = words_and_length.popleft()
            if word == endWord:
                return path_length
            for adjacent_word in adjacents[word]:
                if adjacent_word in visited:
                    continue
                visited.add(adjacent_word)
                words_and_length.append((adjacent_word, path_length + 1))
        return 0
```

### Step3

- 最終的には以下の書き方に落ち着いた
- 隣接する単語の探し方が異なる
- 事前に adjacency list を作るのではなく、毎度隣接する単語を生成して、セットに含まれるかどうかチェックする
- 含まれていたら、wordList に存在する隣接する単語だとわかる


```python
class Solution:
    def ladderLength(self, beginWord: str, endWord: str, wordList: List[str]) -> int:
        candinates = set(wordList)
        if not endWord in candinates:
            return 0
        
        def generate_adjacent_words(word):
            for i in range(len(word)):
                for letter in string.ascii_lowercase:
                    new_word = f'{word[:i]}{letter}{word[i + 1:]}'
                    if not new_word in candinates:
                        continue
                    # 同じ文字が生成されても動く
                    candinates.remove(new_word)
                    yield new_word
        
        path_length = 1
        words = [beginWord]
        while words:
            next_words = []
            for word in words:
                if word == endWord:
                    return path_length
                for adjacent_word in generate_adjacent_words(word):
                    next_words.append(adjacent_word)
            words = next_words
            path_length += 1
        return 0
        
```
