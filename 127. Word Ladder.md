## Memo

### Step1

重みづけがない無向グラフの最短経路を求める問題と理解した。重みづけがない場合の最短経路は、BFSで計算できるのでBFSを選択する。隣接する単語の数が辺の数と等しい。辺の数は多くなさそうなので、グラフは隣接リストで表現する。問題はこの隣接リストをどうやって作るかということ。単純に全てのノードに対して2重ループを回して、ループの中で差をカウントする方法を思いついた。実行には数秒かかりそう。Python なら Leetcode では時間切れ判定になりそう。
num_transformations をカウントしてないことに気付かなくて、結局Geminiに教えてもらって動くようになった。

### Step2

neetcode で解法を確認した。各単語の各文字を記号に置き換えて、それをキーとして元の単語をリストに追加していた。これによって隣接リストを O(NM^2) (N: 単語数、M: 最大の単語の長さ) で作れるようになる。最初は時間計算量はO(N*M)だと思っていたが、記号に置き換える時に文字列をコピーするのでO(NM^2)が正しい。
BFS の時間計算量は一般的なものだと O(V + E) で、これをこの問題に当てはめるとO(NM^2 + NM) => O(NM) になる（各単語にはM個のパターンがあるので、辺の総数はO(NM)になる)。
あと気をつけたいのは、隣接するノードリストに自分自身が含まれていること。訪れたノードを記録すれば弾けるので問題ない。

### Step3

最終的にかかる時間は8分くらいになった。
各文字をアルファベットに置き換える方法と、各文字を記号で置き換える方法で悩んだが、最後は後者を選択した。前者の方が時間計算量が良くて実装が易しいが、単語にアルファベットの小文字しかこないという制約に依存しているのが気になった。実際に記号や大文字が入る英単語は存在する。

class Solution {
public:
    int ladderLength(string beginWord, string endWord, vector<string>& wordList) {
        wordList.push_back(beginWord);
        std::unordered_map<int, std::vector<int>> adjacents;
        for (int i = 0; i < wordList.size(); ++i) {
            for (int j = i + 1; j < wordList.size(); ++j) {
                int difference = 0;
                for (int pos = 0; pos < wordList[i].size(); ++pos) {
                    if (wordList[i][pos] != wordList[j][pos]) {
                        ++difference;
                    }
                }
                if (difference > 1) {
                    continue;
                }
                adjacents[i].push_back(j);
                adjacents[j].push_back(i);
            }
        }
        int begin_word_index = wordList.size() - 1;
        std::vector<int> nodes = {begin_word_index};
        std::unordered_set<int> visited = {begin_word_index};
        int num_transformations = 1;
        while (!nodes.empty()) {
            std::vector<int> next_nodes;
            for (auto node : nodes) {
                for (auto next_node : adjacents[node]) {
                    if (visited.contains(next_node)) {
                        continue;
                    }
                    if (wordList[next_node] == endWord) {
                        return num_transformations + 1;
                    }
                    visited.insert(next_node);
                    next_nodes.push_back(next_node);
                }
            }
            ++num_transformations;
            nodes = std::move(next_nodes);
        }
        return 0;
    }
};

class Solution {
    std::vector<std::string> MakePatterns(const std::string& word) {
        std::vector<std::string> patterns(word.size());
        for (int i = 0; i < word.size(); ++i) {
            std::string pattern = word;
            pattern[i] = '*';
            patterns[i] = std::move(pattern);
        }
        return patterns;
    }

public:
    int ladderLength(string begin_word, string end_word, vector<string>& word_list) {
        std::unordered_map<std::string, std::vector<std::string>> neighbors;
        for (const auto& word : word_list) {
            for (const auto& pattern : MakePatterns(word)) {
                neighbors[pattern].push_back(word);
            }
        }
        std::unordered_set<std::string> visited = {begin_word};
        std::queue<std::pair<std::string, int>> word_and_level;
        word_and_level.emplace(begin_word, 1);
        while (!word_and_level.empty()) {
            auto [word, level] = word_and_level.front();
            word_and_level.pop();
            for (const auto& pattern : MakePatterns(word)) {
                for (const auto& next_word : neighbors[pattern]) {
                    if (next_word == end_word) {
                        return level + 1;
                    }
                    if (visited.contains(next_word)) {
                        continue;
                    }
                    visited.insert(next_word);
                    word_and_level.emplace(next_word, level + 1);
                }
            }
        }
        return 0;
    }
};

class Solution {
public:
    int ladderLength(string beginWord, string endWord, vector<string>& wordList) {
        std::unordered_set<std::string> word_set(
            wordList.begin(), wordList.end()
        );
        std::queue<std::pair<std::string, int>> word_and_count;
        word_and_count.emplace(beginWord, 1);
        while (!word_and_count.empty()) {
            auto [word, count] = std::move(word_and_count.front());
            word_and_count.pop();
            for (int i = 0; i < word.size(); ++i) {
                for (char c = 'a'; c <= 'z'; ++c) {
                    std::string new_word = word;
                    new_word[i] = c;
                    if (!word_set.contains(new_word)) {
                        continue;
                    }
                    if (new_word == endWord) {
                        return count + 1;
                    }
                    word_set.erase(new_word);
                    word_and_count.emplace(new_word, count + 1);
                }
            }
        }
        return 0;
    }
};

class Solution {
    bool DiffersByOneCharacter(std::string word1, std::string word2) {
        if (word1.size() != word2.size()) {
            return false;
        }
        int dfference = 0;
        for (int i = 0; i < word1.size(); ++i) {
            if (word1[i] != word2[i]) {
                ++dfference;
            }
            if (dfference > 1) {
                return false;
            }
        }
        return true;
    }

    void SetNeighbors(const std::vector<std::string>& wordList,
                       std::vector<std::vector<int>>& neighbors) {
        for (int i = 0; i < wordList.size(); ++i) {
            for (int j = i + 1; j < wordList.size(); ++j) {
                if (DiffersByOneCharacter(wordList[i], wordList[j])) {
                    neighbors[i].push_back(j);
                    neighbors[j].push_back(i);
                }
            }
        } 
    }

public:
    int ladderLength(string beginWord, string endWord, vector<string>& wordList) {
        wordList.push_back(beginWord);
        std::vector<std::vector<int>> neighbors(wordList.size());
        SetNeighbors(wordList, neighbors);

        std::vector<uint8_t> visited(wordList.size());
        int begin_word_index = wordList.size() - 1;
        visited[begin_word_index] = 1;
        std::queue<std::pair<int, int>> word_and_level;
        word_and_level.emplace(begin_word_index, 1);
        while (!word_and_level.empty()) {
            auto [word_index, level] = word_and_level.front();
            word_and_level.pop();
            for (auto neighbor_index : neighbors[word_index]) {
                if (wordList[neighbor_index] == endWord) {
                    return level + 1;
                }
                if (visited[neighbor_index]) {
                    continue;
                }
                visited[neighbor_index] = 1;
                word_and_level.emplace(neighbor_index, level + 1);
            }
        }
        return 0;
    }
};

class Solution {
    struct WordAndLevel {
        std::string word;
        int level;
    };

    std::vector<std::string> MakePatterns(const std::string& word) {
        std::vector<std::string> patterns;
        for (int i = 0; i < word.size(); ++i) {
            std::string pattern = word;
            pattern[i] = '*';
            patterns.push_back(std::move(pattern));
        }
        return patterns;
    }

public:
    int ladderLength(string beginWord, string endWord, vector<string>& wordList) {
        std::unordered_map<std::string, std::vector<std::string>> adjacency_list;
        for (const auto& word : wordList) {
            for (const auto& pattern : MakePatterns(word)) {
                adjacency_list[pattern].push_back(word);
            }
        }
        std::unordered_set<std::string> visited = {beginWord};
        std::queue<WordAndLevel> candidates;
        candidates.emplace(beginWord, 1);
        while (!candidates.empty()) {
            auto [word, level] = std::move(candidates.front());
            candidates.pop();
            for (const auto& pattern : MakePatterns(word)) {
                for (const auto& next_word : adjacency_list[pattern]) {
                    if (next_word == endWord) {
                        return level + 1;
                    }
                    if (!visited.contains(next_word)) {
                        visited.insert(next_word);
                        candidates.emplace(next_word, level + 1);
                    }
                }
            }
        }
        return 0;
    }
};
